import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int[] a = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = sc.nextInt();
        }

        boolean triangle = false;
        boolean segment = false;

        for (int i = 0; i < 4; i++) {
            for (int j = i + 1; j < 4; j++) {
                for (int k = j + 1; k < 4; k++) {

                    int[] s = {a[i], a[j], a[k]};
                    Arrays.sort(s);

                    if (s[0] + s[1] > s[2]) {
                        triangle = true;
                    } else if (s[0] + s[1] == s[2]) {
                        segment = true;
                    }
                }
            }
        }

        if (triangle) {
            System.out.println("TRIANGLE");
        } else if (segment) {
            System.out.println("SEGMENT");
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }
}
