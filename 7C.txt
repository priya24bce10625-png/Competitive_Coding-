import java.util.*;

public class Main {

    static long gcd(long a, long b) {
        return b == 0 ? Math.abs(a) : gcd(b, a % b);
    }

    static long[] extendedGCD(long a, long b) {
        if (b == 0) {
            return new long[]{1, 0, a};
        }

        long[] res = extendedGCD(b, a % b);

        long x = res[1];
        long y = res[0] - (a / b) * res[1];

        return new long[]{x, y, res[2]};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        long A = sc.nextLong();
        long B = sc.nextLong();
        long C = sc.nextLong();

        long rhs = -C;

        long g = gcd(A, B);

        if (rhs % g != 0) {
            System.out.println(-1);
            return;
        }

        long[] eg = extendedGCD(Math.abs(A), Math.abs(B));

        long x = eg[0];
        long y = eg[1];

        if (A < 0) x = -x;
        if (B < 0) y = -y;

        long factor = rhs / g;

        x *= factor;
        y *= factor;

        System.out.println(x + " " + y);
    }
}
