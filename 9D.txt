import java.util.*;

public class Main {

    static long[][] dp = new long[36][36];

    static long count(int n, int h) {
        if (n == 0) return 1;
        if (h == 0) return 0;

        if (dp[n][h] != -1) return dp[n][h];

        long res = 0;

        for (int root = 1; root <= n; root++) {
            int left = root - 1;
            int right = n - root;

            res += count(left, h - 1) * count(right, h - 1);
        }

        return dp[n][h] = res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int h = sc.nextInt();

        for (int i = 0; i < 36; i++) {
            Arrays.fill(dp[i], -1);
        }

        long total = count(n, n);
        long lessThanH = count(n, h - 1);

        System.out.println(total - lessThanH);
    }
}
